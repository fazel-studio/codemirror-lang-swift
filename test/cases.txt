# Case 13: switch comprehensive
func testSwitch(x: Int, y: (Int, String), z: String?) {
    switch x {
    case 1:
        print("one")
    case 2, 3:
        print("two or three")
    case 1...5:
        print("range")
    case 1..<10 where x > 5:
        print("half open")
    case let v where v > 10:
        print(v)
    case let v?:
        print(v)
    default:
        print("default")
    }

    switch y {
    case (let a, let b):
        print(a, b)
    case (1, "hi"):
        print("tuple")
    default:
        break
    }

    switch z {
    case let s?:
        print(s)
    case nil:
        print("nil")
    default:
        break
    }

    switch x {
    case 1:
        fallthrough
    case 2:
        print("2 after fallthrough")
    default:
        break
    }
}

# Case 14: if let / guard let optional binding
func testOptionalBinding(a: Int?, b: String?, c: Int) {
    if let x = a {
        print(x)
    }
    if var y = b {
        print(y)
    }
    guard let x = a else { return }
    guard let y = b else { return }

    if let x = a, x > 0, let y = b, y.count > 0 {
        print(x, y)
    }

    if let x = a, let y = b, c > 10 {
        print(x, y, c)
    }

    if case let x? = a {
        print(x)
    }
}
